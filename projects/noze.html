int stroke_weight = 1;

void setup () {
  size(960, 540, P2D);
  background(#000000);
  stroke(#ffffff);
}

void keyPressed() {

  // Cambia el clor de fondo para borrar la pantalla
  if (key == 'c' || key == 'C') {
    background(#000000);
  }

  // Cambia el color 
  // Rojo
  if (key == 'r' || key == 'R') {
    stroke(#ff0000);
  }
  // Verde
  if (key == 'g' || key == 'G') {
    stroke(#00ff00);
  }
  // Azul
  if (key == 'b' || key == 'B') {
    stroke(#0000ff);
  }
  // Amarillo
  if (key == 'y' || key == 'Y') {
    stroke(#ffff00);
  }
  // Magenta
  if (key == 'p' || key == 'P') {
    stroke(#ff00ff);
  }
  // Cian
  if (key == 'a' || key == 'A') {
    stroke(#00ffff);
  }
  // Blanco
  if (key == 'w' || key == 'W') {
    stroke(#ffffff);
  }

  // Cambia el grosor del trazo
  if (key == '1') {
    stroke_weight = 1;
  }
  if (key == '2') {
    stroke_weight = 2;
  }
  if (key == '3') {
    stroke_weight = 3;
  }
  if (key == '4') {
    stroke_weight = 4;
  }
  if (key == '5') {
    stroke_weight = 5;
  }
  if (key == '6') {
    stroke_weight = 6;
  }
  if (key == '7') {
    stroke_weight = 7;
  }
  if (key == '8') {
    stroke_weight = 8;
  }
  if (key == '9') {
    stroke_weight = 9;
  }
  if (key == '0') {
    stroke_weight = 10;
  }
  if (keyCode == UP) {
    stroke_weight += 10;
  }
  if (keyCode == DOWN) {
    stroke_weight -= 10;
  }
  if (keyCode == LEFT) {
    stroke_weight -= 1;
  }
  if (keyCode == RIGHT) {
    stroke_weight += 1;
  }
  
  if (stroke_weight < 1) {
  stroke_weight = 1;
}
  strokeWeight(stroke_weight);

  // Guarda una captura de pantalla
  // Este línea no la he hecho yo la ha sugerido el vscode X copilot
  // Yo solo le he cambiado la path de guardado
  if (key == 's' || key == 'S') {
    saveFrame("./user-imgs/screenshot-######.png");
  }
}
